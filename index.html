  function renderSearchGroupPreview(q, ipg, kat) {

    const box = document.getElementById('searchGroupPreview');

    const grid = document.getElementById('searchGroupGrid');


    if (!q && !ipg && !kat) {

      box.classList.add('hidden');

      return;

    }


    const matchedKumpNames = new Set();

    appData.standings.forEach(s => {

      const catCode = s.kump.split('-')[0];

      if (kat && catCode !== kat) return;

      if (ipg && s.ipg !== ipg) return;

      if (q) {

        const hay = `${s.pasangan} ${s.namaPenuh} ${s.ipg} ${s.kump}`.toLowerCase();

        if (!hay.includes(q)) return;

      }

      matchedKumpNames.add(s.kump);

    });


    if (matchedKumpNames.size === 0) {

      box.classList.add('hidden');

      return;

    }


    box.classList.remove('hidden');

    const kumpList = Array.from(matchedKumpNames).slice(0, 6);


    grid.innerHTML = kumpList.map(kName => {

      const catCode = kName.split('-')[0];

      const members = appData.standings

        .filter(s => s.kump === kName)

        .sort((a, b) => Number(a.kedudukan) - Number(b.kedudukan));


      const rowsHtml = members.map(m => {

        const isMatch = (q && `${m.pasangan} ${m.namaPenuh} ${m.ipg}`.toLowerCase().includes(q)) || (ipg && m.ipg === ipg);

        return `

          <tr class="${isMatch ? 'rr-highlight-row' : ''}">

            <td><b>${m.kedudukan}</b></td>

            <td>${m.pasangan} <span style="color:var(--muted); font-size:0.72rem;">(${m.ipg})</span></td>

            <td>${m.main}</td>

            <td><b>${m.mata}</b></td>

          </tr>

        `;

      }).join('');


      return `

        <div class="rr-mini-card kump-wrap-${catCode}">

          <div class="rr-mini-head kump-head-${catCode}">

            <span>KUMPULAN ${kName}</span>

          </div>

          <table class="rr-mini-table">

            <thead class="kump-subth-${catCode}"><tr><th>#</th><th>Pasangan</th><th>P</th><th>Mt</th></tr></thead>

        </div>

      `;

    }).join('');

  }
